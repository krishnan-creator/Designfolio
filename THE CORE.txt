/**
 * Detect common theme conventions used by apps:
 * - <html data-theme="dark|light">
 * - <html data-mode="dark|light">
 * - Tailwind's .dark class
 * - .light / .theme-light / .theme-dark classes
 * If none are present, fall back to the OS preference.
 * Passing theme="dark" or theme="light" is the most reliable option.
 */
function detectDocumentTheme() {
  if (typeof document === 'undefined') return 'dark';

  const elements = [document.documentElement, document.body].filter(Boolean);

  for (const element of elements) {
    const declaredTheme = `${element.getAttribute('data-theme') || ''} ${element.getAttribute('data-mode') || ''}`.toLowerCase();
    if (declaredTheme.includes('dark')) return 'dark';
    if (declaredTheme.includes('light')) return 'light';
  }

  for (const element of elements) {
    const classes = element.classList;
    if (classes.contains('dark') || classes.contains('theme-dark')) return 'dark';
    if (classes.contains('light') || classes.contains('theme-light')) return 'light';
  }

  const colorScheme = window.getComputedStyle(document.documentElement).colorScheme;
  if (colorScheme?.includes('dark')) return 'dark';
  if (colorScheme?.includes('light')) return 'light';

  return window.matchMedia?.('(prefers-color-scheme: dark)').matches ? 'dark' : 'light';
}

/**
 * Two-column profile section: interactive 3D lanyard on the left,
 * personal context, Tech Arsenal, and a looping career map on the right.
 *
 * theme: "auto" | "dark" | "light". Prefer passing your app's theme state
 * directly (for example theme={isDark ? 'dark' : 'light'}).
 */
export default function LanyardProfileSection({
  theme = 'auto',
  frontImage = '/card-front.png',
  backImage = '/card-back.png',
  strapImage = '/band.png',
  className = '',
}) {
  const [activeTheme, setActiveTheme] = useState(
    theme === 'dark' || theme === 'light' ? theme : 'dark'
  );
  const [selectedStage, setSelectedStage] = useState('CREATIVE DIRECTOR');

  useEffect(() => {
    const syncTheme = () => {
      const nextTheme = theme === 'dark' || theme === 'light' ? theme : detectDocumentTheme();
      setActiveTheme(nextTheme);
    };

    syncTheme();
    if (theme === 'dark' || theme === 'light' || typeof document === 'undefined') {
      return undefined;
    }

    const observer = new MutationObserver(syncTheme);
    const observerOptions = {
      attributes: true,
      attributeFilter: ['class', 'data-theme', 'data-mode', 'style'],
    };
    observer.observe(document.documentElement, observerOptions);
    if (document.body) observer.observe(document.body, observerOptions);

    const media = window.matchMedia?.('(prefers-color-scheme: dark)');
    media?.addEventListener?.('change', syncTheme);

    return () => {
      observer.disconnect();
      media?.removeEventListener?.('change', syncTheme);
    };
  }, [theme]);

  const lanyardColors = activeTheme === 'dark'
    ? {
        cardColor: '#F4F4F1',
        strapColor: '#BFC0C6',
        metal: 'graphite',
      }
    : {
        cardColor: '#FFFFFF',
        strapColor: '#171719',
        metal: 'silver',
      };

  return (
    <section
      className={`lanyard-profile-section ${className}`.trim()}
      data-theme={activeTheme}
      aria-label="C. Bala Krishnan profile and career skill map"
    >
      <div className="lanyard-profile-section__visual">
        <Lanyard
          frontImage={frontImage}
          backImage={backImage}
          strapImage={strapImage}
          imageFit="contain"
          cardColor={lanyardColors.cardColor}
          orientation="portrait"
          finish="matte"
          cornerRadius={0.3}
          size={0.6}
          anchor="left"
          strapLength={0.22}
          strapColor={lanyardColors.strapColor}
          strapWidth={0.6}
          metal={lanyardColors.metal}
          gravity={0.35}
          damping={0.48}
          elasticity={0.2}
          breeze={0.12}
          interactive
          intro
        />
      </div>

      <div className="lanyard-profile-section__content">
        <h1 className="lanyard-profile-section__name">C. BALA KRISHNAN</h1>

        <section className="lanyard-profile-section__block" aria-labelledby="tech-arsenal-title">
          <h2 className="lanyard-profile-section__heading" id="tech-arsenal-title">
            <span className="lanyard-profile-section__index">01 /</span>
            <span>TECH ARSENAL</span>
          </h2>
          <div className="lanyard-profile-section__tech-list">
            {TECH_ARSENAL.map((technology) => (
              <span className="lanyard-profile-section__tech-chip" key={technology}>
                {technology}
              </span>
            ))}
          </div>
        </section>

        <section className="lanyard-profile-section__block" aria-labelledby="career-skill-map-title">
          <h2 className="lanyard-profile-section__heading" id="career-skill-map-title">
            <span className="lanyard-profile-section__index">02 /</span>
            <span>CAREER SKILL MAP</span>
          </h2>

          <p className="lanyard-profile-section__map-instruction">
            Select a stage · Hover or focus to pause
          </p>

          <div className="lanyard-profile-section__career-map">
            <div className="lanyard-profile-section__career-track">
              {[0, 1].map((copy) => (
                <div
                  className="lanyard-profile-section__career-sequence"
                  aria-hidden={copy === 1 ? 'true' : undefined}
                  key={copy}
                >
                  {CAREER_STAGES.map((stage, index) => (
                    <Fragment key={`${copy}-${stage}`}>
                      {copy === 0 ? (
                        <button
                          type="button"
                          className={`lanyard-profile-section__career-stage${selectedStage === stage ? ' is-selected' : ''}`}
                          onClick={() => setSelectedStage(stage)}
                          aria-pressed={selectedStage === stage}
                        >
                          {stage}
                        </button>
                      ) : (
                        <span className={`lanyard-profile-section__career-stage${selectedStage === stage ? ' is-selected' : ''}`}>
                          {stage}
                        </span>
                      )}
                      {index < CAREER_STAGES.length - 1 && (
                        <span className="lanyard-profile-section__career-arrow" aria-hidden="true">→</span>
                      )}
                    </Fragment>
                  ))}
                  <span className="lanyard-profile-section__career-arrow" aria-hidden="true">→</span>
                </div>
              ))}
            </div>
          </div>

          <p className="lanyard-profile-section__tagline">The journey behind the practice.</p>
        </section>
      </div>
    </section>
  );
}

# Lanyard Profile Section

This is a two-column React section designed to sit inside a portfolio page:

- Left: the existing ReactBits `Lanyard` component.
- Right: C. Bala Krishnan, the Tech Arsenal chips, and an infinitely looping, interactive Career Skill Map.
- Dark/light: the wrapper detects common `data-theme`, `data-mode`, `.dark`, and `.light` conventions and changes section styling plus Lanyard card/strap/metal colors.
- Career map: loops continuously, pauses on hover/focus, and the first row's stages are clickable. Reduced-motion preferences disable the animation.

## Files

- `LanyardProfileSection.jsx`
- `LanyardProfileSection.css`

Keep these beside your existing `Lanyard.jsx` / `Lanyard.css`, or update the import path to match your project.

## Usage

```jsx
import LanyardProfileSection from './LanyardProfileSection';

export default function AboutSection({ isDark }) {
  return (
    <LanyardProfileSection
      theme={isDark ? 'dark' : 'light'}
      frontImage="/card-front.png"
      backImage="/card-back.png"
      strapImage="/band.png"
    />
  );
}
```

If your app's active theme is represented on `html` or `body` via `data-theme="dark|light"`, `data-mode="dark|light"`, `.dark`, or `.light`, you can omit `theme` or set `theme="auto"`. Passing the app's actual theme state directly is the most reliable option.

Make sure the image paths point to real files in your app's `public` directory, or pass the correct asset URLs. The component code expects the existing `Lanyard` implementation and the `three` package to be installed.
